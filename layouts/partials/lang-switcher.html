{{- /* Custom 4-language switcher — links to the same page in each language. zh = default language, no URL prefix. en/ja/vi use /en/, /ja/, /vi/ prefixes directly after the site base path (works with or without a subpath like /tech-blog/). Works on homepage, posts, and section pages by swapping the language prefix segment. */ -}} {{- $parsed := urls.Parse .Site.BaseURL -}} {{- $basePath := strings.TrimSuffix "/" $parsed.Path -}} {{- $base := printf "https://%s" $parsed.Host -}} {{- $rel := .RelPermalink -}} {{- $after := strings.TrimPrefix $basePath $rel -}} {{- /* detect current language from URL prefix */ -}} {{- $cur := "zh" -}} {{- range slice "en" "ja" "vi" -}} {{- if strings.HasPrefix $after (printf "/%s/" .) }}{{ $cur = . }}{{ end -}} {{- end -}} {{- /* strip current language prefix to get the language-neutral path */ -}} {{- $stripped := $after -}} {{- if ne $cur "zh" }}{{ $stripped = replace $after (printf "/%s/" $cur) "/" 1 }}{{ end -}} {{- $zhUrl := printf "%s%s%s" $base $basePath $stripped -}} {{- $enUrl := printf "%s%s/en%s" $base $basePath $stripped -}} {{- $jaUrl := printf "%s%s/ja%s" $base $basePath $stripped -}} {{- $viUrl := printf "%s%s/vi%s" $base $basePath $stripped -}}
