{{- /* Custom 4-language switcher — links to the same page in each language. Uses Hugo's .Translations so the link always points at the real translated page. When a translation is missing, falls back to that language's home page — never generates a 404 link. zh = default language, no URL prefix. en/ja/vi use /en/, /ja/, /vi/. */ -}} {{- $pg := . -}} {{- $cur := .Site.Language.Lang -}} {{- $langs := slice "zh" "en" "ja" "vi" -}} {{- $labels := dict "zh" "中" "en" "EN" "ja" "日" "vi" "VI" -}} {{- $arias := dict "zh" "中文" "en" "English" "ja" "日本語" "vi" "Tiếng Việt" -}}
{{- range $l := $langs }} {{- $href := "" -}} {{- if eq $l $cur -}} {{- $href = $pg.RelPermalink -}} {{- else -}} {{- range $tr := $pg.Translations -}} {{- if eq $tr.Lang $l -}}{{- $href = $tr.RelPermalink -}}{{- end -}} {{- end -}} {{- if eq $href "" -}} {{- /* no translation for this page: fall back to that language's home */ -}} {{- range $s := $pg.Sites -}} {{- if eq $s.Language.Lang $l -}}{{- $href = $s.Home.RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- end }} {{ index $labels $l }} {{- end }}
